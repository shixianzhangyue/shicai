```rust
use serde::{Deserialize, Serialize};

#[derive(Debug, Serialize, Deserialize, Clone)]
pub struct BaiduOcrConfig {
    pub api_key: String,
    pub secret_key: String,
    pub access_token: Option<String>,
    pub token_expires_at: Option<i64>,
}

#[derive(Debug, Serialize)]
pub struct OcrResult {
    pub text: String,
    pub confidence: f64,
    pub words: Vec<WordItem>,
}

#[derive(Debug, Serialize)]
pub struct WordItem {
    pub text: String,
    pub confidence: f64,
    pub location: Location,
}

#[derive(Debug, Serialize)]
pub struct Location {
    pub top: i32,
    pub left: i32,
    pub width: i32,
    pub height: i32,
}

impl BaiduOcrConfig {
    /// 获取 access_token（有效期 30 天，提前 1 天刷新）
    pub async fn ensure_token(&mut self) -> Result<String, String> {
        // 检查缓存 token 是否有效
        if let (Some(token), Some(expires)) = (&self.access_token, self.token_expires_at) {
            let now = chrono::Utc::now().timestamp();
            if now < expires - 86400 {
                return Ok(token.clone());
            }
        }

        // 请求新 token
        let url = format!(
            "https://aip.baidubce.com/oauth/2.0/token?grant_type=client_credentials&client_id={}&client_secret={}",
            self.api_key, self.secret_key
        );

        let client = reqwest::Client::new();
        let resp = client.get(&url).send().await.map_err(|e| e.to_string())?;
        let json: serde_json::Value = resp.json().await.map_err(|e| e.to_string())?;

        let token = json["access_token"]
            .as_str()
            .ok_or("获取 token 失败：响应中无 access_token")?
            .to_string();

        self.access_token = Some(token.clone());
        self.token_expires_at = Some(chrono::Utc::now().timestamp() + 2592000);

        Ok(token)
    }

    /// 通用文字识别（高精度版）
    pub async fn recognize_general(&mut self, image_base64: &str) -> Result<OcrResult, String> {
        let token = self.ensure_token().await?;
        let url = format!(
            "https://aip.baidubce.com/rest/2.0/ocr/v1/accurate_basic?access_token={}",
            token
        );

        let client = reqwest::Client::new();
        let params = [
            ("image", image_base64),
            ("language_type", "CHN_ENG"),
            ("detect_direction", "true"),
        ];

        let resp = client
            .post(&url)
            .form(&params)
            .send()
            .await
            .map_err(|e| e.to_string())?;

        let json: serde_json::Value = resp.json().await.map_err(|e| e.to_string())?;

        // 解析返回结果
        let words = json["words_result"]
            .as_array()
            .unwrap_or(&vec![])
            .iter()
            .map(|w| WordItem {
                text: w["words"].as_str().unwrap_or("").to_string(),
                confidence: w["probability"]["average"].as_f64().unwrap_or(0.0),
                location: Location {
                    top: w["location"]["top"].as_i64().unwrap_or(0) as i32,
                    left: w["location"]["left"].as_i64().unwrap_or(0) as i32,
                    width: w["location"]["width"].as_i64().unwrap_or(0) as i32,
                    height: w["location"]["height"].as_i64().unwrap_or(0) as i32,
                },
            })
            .collect();

        let text = words.iter().map(|w| w.text.as_str()).collect::<Vec<_>>().join("\n");
        let confidence = json["words_result_num"].as_f64().unwrap_or(0.0);

        Ok(OcrResult { text, confidence, words })
    }
}
```

#### C3：OCR 解析命令

**新增文件：** `src-tauri/src/commands/ocr_parser.rs`

```rust
use crate::db::pool::DbPool;
use crate::services::baidu_ocr::BaiduOcrConfig;
use std::sync::Mutex;

/// 使用百度 OCR 解析简历
#[tauri::command]
pub async fn parse_resume_ocr(
    state: tauri::State<'_, DbPool>,
    ocr_config: tauri::State<'_, Mutex<BaiduOcrConfig>>,
    resume_id: String,
) -> Result<serde_json::Value, String> {
    let conn = state.get().map_err(|e| e.to_string())?;

    // 1. 获取简历文件路径
    let file_path: String = conn
        .query_row(
            "SELECT file_path FROM resumes WHERE id = ?1",
            [&resume_id],
            |row| row.get(0),
        )
        .map_err(|e| format!("简历不存在: {}", e))?;

    // 2. 更新状态为 parsing
    conn.execute(
        "UPDATE resumes SET parse_status = 'parsing', parse_method = 'ocr' WHERE id = ?1",
        [&resume_id],
    ).map_err(|e| e.to_string())?;

    // 3. 读取文件并转为 base64
    let file_bytes = std::fs::read(&file_path).map_err(|e| e.to_string())?;
    let image_base64 = base64::encode(&file_bytes);

    // 4. 调用 OCR
    let mut config = ocr_config.lock().map_err(|e| e.to_string())?;
    let ocr_result = config.recognize_general(&image_base64).await?;

    // 5. 存储原始 OCR 结果
    let ocr_json = serde_json::to_string(&ocr_result).map_err(|e| e.to_string())?;
    conn.execute(
        "UPDATE resumes SET raw_ocr_result = ?1, parse_status = 'completed' WHERE id = ?2",
        [&ocr_json, &resume_id],
    ).map_err(|e| e.to_string())?;

    // 6. 提取结构化数据
    let parsed = extract_structured_data(&ocr_result.text);

    Ok(parsed)
}

/// 从 OCR 文本提取结构化简历数据
fn extract_structured_data(text: &str) -> serde_json::Value {
    // 简单的正则提取逻辑
    let name = extract_field(text, &["姓名", "名字"]);
    let phone = extract_phone(text);
    let email = extract_email(text);
    let education = extract_education(text);

    serde_json::json!({
        "name": name,
        "phone": phone,
        "email": email,
        "education": education,
        "rawText": text,
    })
}
```

## 八、风险与注意事项

### 8.1 技术风险

| 风险 | 影响 | 缓解措施 |
|------|------|---------|
| 百度 OCR API 配额限制 | 解析次数受限 | 实现 token 缓存 + 错误重试 |
| PDF 转图片兼容性 | 部分 PDF 无法转换 | 降级为直接调用系统阅读器 |
| 大文件 OCR 超时 | 解析失败 | 实现异步解析 + 进度通知 |
| 数据库迁移冲突 | 迁移失败 | 严格按版本号顺序执行 |

### 8.2 开发规范

1. **数据库迁移**：禁止修改已发布的迁移文件，只能新增
2. **软删除**：所有 jobs/candidates 查询必须过滤 `deleted_at IS NULL`
3. **输入校验**：所有 Tauri command 使用 `validate.rs` 中的 Input struct
4. **错误处理**：使用 `map_err` + `log::error!` 模式
5. **日志记录**：ERROR 级别异常必须记录到文件日志
6. **API 密钥**：百度 OCR AK/SK 通过设置页配置，不硬编码

### 8.3 向后兼容

- 保留 `candidates` 表结构，仅新增字段
- 保留 `jobs` 表结构，仅新增 `headcount` 字段
- 保留现有 API 接口，仅扩展参数
- 删除流程模板前确保无数据依赖
